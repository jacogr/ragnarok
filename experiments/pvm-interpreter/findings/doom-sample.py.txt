"""Sample a fully checked unmodified DOOM image separately; discard every profiled clock."""
from pathlib import Path
import hashlib,json,os,platform,re,subprocess,time
R=Path.cwd();B=R/'build/pvm-interpreter-experiment/doom-sample';B.mkdir(exist_ok=True)
C=R/'build/pvm-interpreter-experiment/comparisons/binary-checkpoint'
d=json.loads((C/'results.json').read_text());assert d['status']=='passed' and d['all_inputs_stable']
exe=C/'candidate-doom';before=hashlib.sha256(exe.read_bytes()).hexdigest()
env=dict(os.environ,RAGNAROK_PVM_DOOM_INPUT=str(R/'build/pvm-port-lab/large-app-qualification/prepared'),RAGNAROK_PVM_DOOM_ROM=str(R/'vendor/polkavm/examples/doom/roms/doom1.wad'),RAGNAROK_PVM_DOOM_PHASE='init',RAGNAROK_PVM_DOOM_TIMING='1',RAGNAROK_PVM_LOCAL_ONECALL='0')
with (B/'run.log').open('w') as log:
 p=subprocess.Popen([str(exe)],env=env,stdout=log,stderr=subprocess.STDOUT)
 try:
  # Let input construction finish; the sampled image retains all normal assertion checks.
  time.sleep(2)
  assert p.poll() is None,'Image finished before sampling'
  sampled=subprocess.run(['/usr/bin/sample',str(p.pid),'3','1','-file',str(B/'sample.txt')],capture_output=True,text=True,timeout=15)
  (B/'sampler.log').write_text(sampled.stdout+sampled.stderr);assert sampled.returncode==0
  assert p.wait(timeout=40)==0
 finally:
  if p.poll() is None:p.kill();p.wait()
text=(B/'run.log').read_text();assert 'DOOM_OK,prepared,init' in text and text.count('DOOM_SAMPLE,prepared,init,')==13
assert hashlib.sha256(exe.read_bytes()).hexdigest()==before
sample=(B/'sample.txt').read_text();section=sample.split('Sort by top of stack, same collapsed (when >= 5):',1)[1].split('Binary Images:',1)[0]
leaves=[]
for line in section.splitlines():
 m=re.match(r'\s*(.+?)\s+\(in (.+?)\)\s+(\d+)\s*$',line)
 if m:leaves.append(dict(symbol=m[1],image=m[2],hits=int(m[3])))
proof=dict(status='passed',comparison='binary-checkpoint',core_sha256=d['source_sha256']['experiments/pvm-interpreter/overlay/src/pvm/experimental.nim'],image_sha256=before,profile_sha256=hashlib.sha256((B/'sample.txt').read_bytes()).hexdigest(),architecture=platform.machine(),policy='Statistical sample hits are not CPU percentages, instruction counts or hardware misses. All profiled clocks discarded.',checked_calls=13,delay_seconds=2,duration_seconds=3,interval_ms=1,collapsed_leaves=leaves)
(B/'results.json').write_text(json.dumps(proof,indent=2)+'\n');print(json.dumps(dict(status='passed',top=leaves[:15])))
