"""Count executed register-first pairs in a private fully checked DOOM image; discard all clocks."""
from pathlib import Path
import hashlib,json,os,shutil,sys
R=Path.cwd();S=R/'experiments/pvm-interpreter';B=R/'build/pvm-interpreter-experiment/doom-pair-probe'
B.mkdir(exist_ok=True);sys.path.insert(0,str(R/'tools'))
from native_checks import NativeChecks
from setup_nimony import installed
shutil.copytree(R/'src',B/'src',dirs_exist_ok=True)
shutil.copy2(S/'overlay/src/pvm/machine.nim',B/'src/pvm/machine.nim')
s=(S/'overlay/src/pvm/experimental.nim').read_text();original=hashlib.sha256(s.encode()).hexdigest()
needle='    dirtyPages: seq[uint64] # One bounded 128 KiB bitmap covers all 32-bit guest pages.'
assert s.count(needle)==1;s=s.replace(needle,needle+'\n    probePairs: seq[uint64] # Diagnostic only; all clocks are discarded.')
needle='    dirtyPages: newSeq[uint64](16384),';assert s.count(needle)==1
s=s.replace(needle,needle+' probePairs: newSeq[uint64](65536),')
for name,key in [('flatRegisterPair[Code: static[int]]','Code'),('flatFusedRegisters[First, Second: static[int]]','First')]:
 start=s.index('proc '+name);at=s.index('  let a = f.arguments[int(off)]',start)
 s=s[:at]+'''  if f.barrier != off+1:
    # Count only pairs whose second source boundary is actually attempted.
    let second = int(f.plan.facts[int(f.sites[int(off+1)])].opcode)
    inc f.probePairs['''+key+'''*256+second]
'''+s[at:]
s+='\nfunc flatPairProbe*(f:FlatFrame):seq[uint64] =\n  f.probePairs\n'
(B/'src/pvm/experimental.nim').write_text(s);(B/'benchmarks').mkdir(exist_ok=True)
s=(S/'benchmarks/doom.nim').read_text();needle='  let stats = flatHostStats(frame)';assert s.count(needle)==1
s=s.replace(needle,'''  let pairs = flatPairProbe(frame)
  for i in 0..<pairs.len:
    # Export every nonzero pair, including non-register successors, without selecting timing samples.
    if pairs[i]>0:echo "DOOM_PAIR,",phase,",",i div 256,",",i mod 256,",",pairs[i]
'''+needle)
(B/'benchmarks/doom.nim').write_text(s)
checks=NativeChecks(B);build,exe=checks.build(installed()[0],B/'benchmarks/doom.nim','pair-probe',cflags=['-I'+str(R/'tests/performance')]);assert build['exit_code']==0,build['output'][-1600:]
rows=[]
for phase in ['init','tick']:
 env=dict(os.environ,RAGNAROK_PVM_DOOM_INPUT=str(R/'build/pvm-port-lab/large-app-qualification/prepared'),RAGNAROK_PVM_DOOM_ROM=str(R/'vendor/polkavm/examples/doom/roms/doom1.wad'),RAGNAROK_PVM_DOOM_PHASE=phase,RAGNAROK_PVM_DOOM_TIMING='0',RAGNAROK_PVM_LOCAL_ONECALL='1')
 run=checks.command([exe],phase+'-run',environment=env);assert run['exit_code']==0 and 'DOOM_OK,' in run['output'],run['output'][-1800:]
 counts=[list(map(int,line.split(',')[2:])) for line in run['output'].splitlines() if line.startswith('DOOM_PAIR,')];counts.sort(key=lambda x:x[2],reverse=True)
 print(phase,'top pairs',counts[:15],flush=True);rows.append(dict(phase=phase,counts=counts,run=run))
(B/'results.json').write_text(json.dumps(dict(status='passed',base_core_sha256=original,build=build,rows=rows,clock_policy='All diagnostic clocks discarded'),indent=2)+'\n')
