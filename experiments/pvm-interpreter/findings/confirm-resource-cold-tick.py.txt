"""One predeclared four-pair cold-gameplay confirmation; retain the two originals and every new sample."""
import json,shutil,sys
from pathlib import Path
from types import SimpleNamespace
sys.path.insert(0,str(Path('experiments/pvm-interpreter').resolve()))
from compare import Comparison,ROOT
folder=ROOT/'build/pvm-interpreter-experiment/comparisons/resource-bounds'
data=json.loads((folder/'results.json').read_text())
assert data['status']=='passed' and not data.get('confirmation_plan')
large=ROOT/'build/pvm-port-lab/large-app-qualification'
subject=Comparison(SimpleNamespace(name='resource-bounds',deadline='2026-10-27T03:35:00Z',baseline=data['baseline'],doom_fixtures=large/'prepared',rom=ROOT/'vendor/polkavm/examples/doom/roms/doom1.wad'))
subject.report=data;subject.verify()
shutil.copy2(folder/'results.json',folder/'results-before-cold-tick-confirmation.json')
subject.report.update(status='confirming',all_inputs_stable=False,confirmation_plan=dict(cold_tick_additional_pairs=4,original_pairs_retained=True,reason='Original cold gameplay pairs have a wide spread; exactly four additional pairs are declared before execution.'))
subject.seal([Path(__file__).resolve()])
try:
    for order in range(2,6):
        # Each adjacent pair shares fixed inputs/images, alternating which role executes first.
        roles=['baseline','candidate']
        for role in roles if order%2==0 else reversed(roles):subject.record(subject.doom(role,'tick',order,cold=True))
        print('PASS additional cold gameplay pair',order,flush=True)
    subject.verify();subject.report.update(status='passed',all_inputs_stable=True)
except (OSError,ValueError,RuntimeError) as error:
    subject.report.update(status='failed',error=str(error));raise
finally:subject.save()
