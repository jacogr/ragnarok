"""Verify actual composed-helper call removal before timing; static code is not a hardware metric."""
from pathlib import Path
import hashlib,json,re,subprocess
B=Path('build/pvm-interpreter-experiment/comparisons/fixed-expanded');exe=B/'candidate-doom'
s=subprocess.check_output(['otool','-tvV',str(exe)],text=True);(B/'native-disassembly.txt').write_text(s)
p=re.split(r'(?m)^(_[^\n]+):\n',s);rows=[]
for i in range(1,len(p),2):
 name,body=p[i:i+2]
 if name.startswith('_flatChargedShape_') or name.startswith('_flatCharge_'):
  calls=[l.strip() for l in body.splitlines() if re.search(r'\b(bl|blr|br)\s',l)]
  assert not any(re.search(r'flatReserve|flatFusedRegisters|flatRegister_|flatBranch_|branchConditionOperands',l) for l in calls),(name,calls)
  rows.append(dict(symbol=name,disassembled_lines=len(body.splitlines()),calls=calls))
assert len([r for r in rows if r['symbol'].startswith('_flatChargedShape_')])==3
assert len([r for r in rows if r['symbol'].startswith('_flatCharge_')])==1
(B/'native-call-proof.json').write_text(json.dumps(dict(status='passed',image_sha256=hashlib.sha256(exe.read_bytes()).hexdigest(),policy='Static ARM64 code facts only; no executed instruction counts or CPU cause inferred',rows=rows),indent=2)+'\n')
print('PASS actual ARM64: all3charged shapes and ordinary charge omit the composed reserve/register calls',flush=True)
