"""Compact structured evidence without changing measurements; keep duplicate full command logs in ignored build storage."""
from pathlib import Path
import hashlib,json,shutil
R=Path.cwd();S=R/'experiments/pvm-interpreter/findings';B=R/'build/pvm-interpreter-experiment/evidence-originals';B.mkdir(exist_ok=True)

def numbers(v,path=()):
    """Bind every original numeric/bool/null field to its unchanged JSON coordinate."""
    out={}
    if isinstance(v,dict):
        for k,x in v.items():out.update(numbers(x,path+(k,)))
    elif isinstance(v,list):
        for i,x in enumerate(v):out.update(numbers(x,path+(i,)))
    elif v is None or isinstance(v,(int,float,bool)):out[path]=v
    return out

def logs(v):
    """Only duplicate textual command output is replaced by a hash/bounded excerpt; parsed counters remain complete."""
    if isinstance(v,dict):
        if all(k in v for k in ['command','exit_code','log','output']) and isinstance(v['output'],str):
            raw=v.pop('output');v['output_sha256']=hashlib.sha256(raw.encode()).hexdigest()
            v['output_excerpt']=raw if len(raw)<=800 else raw[:390]+'\n[Full command output retained in ignored log]\n'+raw[-390:]
            v['output_chars']=len(raw)
            p=R/v['log']
            if p.is_file() and p.read_text(errors='replace')==raw:v['log_sha256']=hashlib.sha256(p.read_bytes()).hexdigest()
        for x in v.values():logs(x)
    elif isinstance(v,list):
        for x in v:logs(x)

def render(v,level=0):
    """Keep small records/numeric rows on one line while retaining ordinary readable JSON containers."""
    packed=json.dumps(v,ensure_ascii=True,separators=(', ',': '))
    if not isinstance(v,(dict,list)) or len(packed)+2*level<=120:return packed
    indent='  '*level;child='  '*(level+1)
    if isinstance(v,dict):
        return '{\n'+',\n'.join(child+json.dumps(k)+': '+render(x,level+1) for k,x in v.items())+'\n'+indent+'}'
    if v and all(x is None or isinstance(x,(int,float,bool)) for x in v):
        rows=[];line=child
        for i,x in enumerate(v):
            token=json.dumps(x)+(',' if i+1<len(v) else '')
            if len(line)+len(token)+1>120 and line!=child:rows.append(line.rstrip());line=child
            line+=token+' '
        rows.append(line.rstrip());return '[\n'+'\n'.join(rows)+'\n'+indent+']'
    return '[\n'+',\n'.join(child+render(x,level+1) for x in v)+'\n'+indent+']'

records=[]
for p in sorted(S.glob('*.json')):
    before=p.read_bytes();original=json.loads(before);old_numbers=numbers(original)
    v=json.loads(before);logs(v);after=(render(v)+'\n').encode();parsed=json.loads(after);assert parsed==v
    current=numbers(parsed);assert all(k in current and current[k]==x for k,x in old_numbers.items()),p.name
    if after!=before:
        backup=B/p.name
        if not backup.exists():shutil.copy2(p,backup)
        p.write_bytes(after)
    records.append(dict(file=p.name,original_sha256=hashlib.sha256(before).hexdigest(),compact_sha256=hashlib.sha256(after).hexdigest(),original_bytes=len(before),compact_bytes=len(after)))
(B/'manifest.json').write_text(json.dumps(records,indent=2)+'\n')
print('PASS compact evidence',len(records),'files; numeric fields unchanged; bytes',sum(r['original_bytes'] for r in records),'->',sum(r['compact_bytes'] for r in records))
