"""Check actual fast-store calls disappeared while the checked slow store remains referenced."""
from pathlib import Path
import hashlib,json,re,subprocess
B=Path('build/pvm-interpreter-experiment/comparisons/store-splice');rows=[]
for driver in ['kernel','lifecycle','doom']:
 exe=B/('candidate-'+driver);text=subprocess.check_output(['otool','-tvV',str(exe)],text=True)
 (B/(driver+'-native-disassembly.txt')).write_text(text)
 fast=[line.strip() for line in text.splitlines() if re.search(r'\b(?:bl|b)\s+_flatStore_\w+',line)]
 slow=[line.strip() for line in text.splitlines() if re.search(r'\b(?:bl|b)\s+_flatStoreSlow_\w+',line)]
 assert not fast,(driver,fast[:8]);assert slow,driver
 rows.append(dict(driver=driver,image_sha256=hashlib.sha256(exe.read_bytes()).hexdigest(),fast_store_direct_calls=fast,checked_slow_store_call_sites=len(slow)))
(B/'native-store-proof.json').write_text(json.dumps(dict(status='passed',policy='Static code facts only, not executed instruction counts or a CPU cause attribution',rows=rows),indent=2)+'\n');print('PASS actual3driver fast-store call removal; checked slow fallback remains',flush=True)
